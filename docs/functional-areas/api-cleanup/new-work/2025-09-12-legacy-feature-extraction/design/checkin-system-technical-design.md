## Executive Summary

This technical design document specifies the implementation of the CheckIn System for WitchCityRope event attendee management. The system follows the established vertical slice architecture pattern and provides mobile-first, offline-capable check-in functionality with <1 second response times.

**AUTHENTICATION APPROACH**: Kiosk Mode with cryptographic session tokens generated by admins. No user authentication required for volunteers operating check-in screens.

## Business Context & Requirements

### Problem Statement
Event organizers need efficient check-in management for community events but currently lack:
- Mobile-optimized interface for volunteer staff
- **Simple handoff process** where admins don't need to share credentials
- Offline capability for unreliable venue WiFi
- Integration with attendee registration and capacity management
- Real-time attendance tracking with safety information display

### Success Metrics
- **Performance**: Check-in processing time <1 second locally, <3 seconds with sync
- **Reliability**: 99%+ offline sync success rate
- **Usability**: <10 minutes volunteer training time (no login required)
- **Capacity**: Support 500+ attendee events
- **Offline**: 4+ hours operation without connectivity
- **Security**: Zero unauthorized admin access incidents

## Authentication Architecture - KIOSK MODE

### Session Token System

**How It Works**:
1. Admin logs into admin panel with full credentials
2. Admin navigates to event and clicks "Generate Check-In Link"
3. System creates cryptographic session token tied to specific event
4. Admin receives URL: `/checkin?session={token}&event={eventId}`
5. Admin opens link on volunteer device (or hands link to volunteer)
6. Check-in screen operates in kiosk mode - **NO LOGIN REQUIRED**
7. All API calls include session token in custom header `X-CheckIn-Token`
8. Token expires after 12 hours or event end (whichever comes first)

### SessionTokenService Implementation

```csharp
/// <summary>
/// Session token service for kiosk mode authentication
/// Implements cryptographically secure token generation and validation
/// </summary>
public class SessionTokenService : ISessionTokenService
{
    private readonly ApplicationDbContext _context;
    private readonly ILogger<SessionTokenService> _logger;
    private readonly IMemoryCache _cache;

    // Token configuration
    private const int TOKEN_LENGTH = 64; // 512 bits
    private const string TOKEN_CACHE_PREFIX = "checkin_token_";

    public async Task<Result<SessionTokenResponse>> GenerateSessionTokenAsync(
        Guid eventId,
        Guid adminUserId,
        int expirationHours = 12,
        CancellationToken cancellationToken = default)
    {
        try
        {
            // Validate event exists
            var eventExists = await _context.Events
                .AnyAsync(e => e.Id == eventId, cancellationToken);

            if (!eventExists)
            {
                return Result<SessionTokenResponse>.Failure("Event not found");
            }

            // Generate cryptographically secure token
            var tokenBytes = new byte[TOKEN_LENGTH];
            using (var rng = RandomNumberGenerator.Create())
            {
                rng.GetBytes(tokenBytes);
            }
            var token = Convert.ToBase64String(tokenBytes)
                .Replace("+", "-")
                .Replace("/", "_")
                .Replace("=", "");

            // Calculate expiration
            var expiresAt = DateTime.UtcNow.AddHours(expirationHours);

            // Store token in database
            var sessionToken = new CheckInSessionToken
            {
                Id = Guid.NewGuid(),
                Token = token,
                EventId = eventId,
                GeneratedBy = adminUserId,
                GeneratedAt = DateTime.UtcNow,
                ExpiresAt = expiresAt,
                IsRevoked = false,
                IsActive = true
            };

            _context.CheckInSessionTokens.Add(sessionToken);
            await _context.SaveChangesAsync(cancellationToken);

            // Cache token for fast validation (12 hour TTL)
            var cacheKey = $"{TOKEN_CACHE_PREFIX}{token}";
            _cache.Set(cacheKey, sessionToken, TimeSpan.FromHours(expirationHours));

            _logger.LogInformation(
                "Generated check-in session token for event {EventId} by admin {AdminId}, expires {ExpiresAt}",
                eventId, adminUserId, expiresAt);

            // Generate check-in URL
            var checkInUrl = $"/checkin?session={token}&event={eventId}";

            return Result<SessionTokenResponse>.Success(new SessionTokenResponse
            {
                Token = token,
                EventId = eventId,
                CheckInUrl = checkInUrl,
                ExpiresAt = expiresAt,
                GeneratedBy = adminUserId,
                GeneratedAt = DateTime.UtcNow
            });
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error generating session token for event {EventId}", eventId);
            return Result<SessionTokenResponse>.Failure("Failed to generate session token");
        }
    }

    public async Task<Result<TokenValidationResult>> ValidateTokenAsync(
        string token,
        Guid requestedEventId,
        CancellationToken cancellationToken = default)
    {
        try
        {
            // Check cache first for performance
            var cacheKey = $"{TOKEN_CACHE_PREFIX}{token}";
            if (_cache.TryGetValue(cacheKey, out CheckInSessionToken cachedToken))
            {
                // Validate cached token
                if (ValidateTokenRules(cachedToken, requestedEventId))
                {
                    return Result<TokenValidationResult>.Success(new TokenValidationResult
                    {
                        IsValid = true,
                        EventId = cachedToken.EventId,
                        ExpiresAt = cachedToken.ExpiresAt
                    });
                }
            }

            // Cache miss or invalid - check database
            var sessionToken = await _context.CheckInSessionTokens
                .FirstOrDefaultAsync(t => t.Token == token, cancellationToken);

            if (sessionToken == null)
            {
                _logger.LogWarning("Invalid check-in token attempted: {Token}", token.Substring(0, 10));
                return Result<TokenValidationResult>.Failure("Invalid session token");
            }

            // Validate token rules
            if (!ValidateTokenRules(sessionToken, requestedEventId))
            {
                return Result<TokenValidationResult>.Failure("Session token validation failed");
            }

            // Update last used timestamp
            sessionToken.LastUsedAt = DateTime.UtcNow;
            await _context.SaveChangesAsync(cancellationToken);

            // Refresh cache
            _cache.Set(cacheKey, sessionToken, sessionToken.ExpiresAt - DateTime.UtcNow);

            return Result<TokenValidationResult>.Success(new TokenValidationResult
            {
                IsValid = true,
                EventId = sessionToken.EventId,
                ExpiresAt = sessionToken.ExpiresAt
            });
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error validating session token");
            return Result<TokenValidationResult>.Failure("Token validation error");
        }
    }

    private bool ValidateTokenRules(CheckInSessionToken token, Guid requestedEventId)
    {
        // Rule 1: Token must not be revoked
        if (token.IsRevoked || !token.IsActive)
        {
            _logger.LogWarning("Revoked token attempted for event {EventId}", requestedEventId);
            return false;
        }

        // Rule 2: Token must not be expired
        if (token.ExpiresAt < DateTime.UtcNow)
        {
            _logger.LogWarning("Expired token attempted for event {EventId}", requestedEventId);
            return false;
        }

        // Rule 3: Token must match requested event
        if (token.EventId != requestedEventId)
        {
            _logger.LogWarning(
                "Token event mismatch: token for {TokenEventId}, requested {RequestedEventId}",
                token.EventId, requestedEventId);
            return false;
        }

        return true;
    }

    public async Task<Result<bool>> RevokeTokenAsync(
        string token,
        Guid adminUserId,
        string reason,
        CancellationToken cancellationToken = default)
    {
        try
        {
            var sessionToken = await _context.CheckInSessionTokens
                .FirstOrDefaultAsync(t => t.Token == token, cancellationToken);

            if (sessionToken == null)
            {
                return Result<bool>.Failure("Token not found");
            }

            // Revoke token
            sessionToken.IsRevoked = true;
            sessionToken.IsActive = false;
            sessionToken.RevokedAt = DateTime.UtcNow;
            sessionToken.RevokedBy = adminUserId;
            sessionToken.RevocationReason = reason;

            await _context.SaveChangesAsync(cancellationToken);

            // Remove from cache
            var cacheKey = $"{TOKEN_CACHE_PREFIX}{token}";
            _cache.Remove(cacheKey);

            _logger.LogInformation(
                "Session token revoked for event {EventId} by admin {AdminId}, reason: {Reason}",
                sessionToken.EventId, adminUserId, reason);

            return Result<bool>.Success(true);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error revoking session token");
            return Result<bool>.Failure("Failed to revoke token");
        }
    }
}
```

```csharp
/// <summary>
/// Admin endpoints for managing check-in sessions

public static class AdminCheckInEndpoints
{
    public static void MapAdminCheckInEndpoints(this IEndpointRouteBuilder app)
    {
        var group = app.MapGroup("/api/admin/checkin")
            .WithTags("Admin CheckIn")
            .RequireAuthorization(policy => policy.RequireRole("Administrator"));

        // Generate session token for event
        group.MapPost("/events/{eventId}/generate-session", GenerateSessionToken)
            .WithName("GenerateCheckInSession")
            .WithOpenApi();

        // Revoke active session token
        group.MapPost("/sessions/{token}/revoke", RevokeSessionToken)
            .WithName("RevokeCheckInSession")
            .WithOpenApi();

        // Get active sessions for event
        group.MapGet("/events/{eventId}/active-sessions", GetActiveSessions)
            .WithName("GetActiveCheckInSessions")
            .WithOpenApi();
    }

    private static async Task<IResult> GenerateSessionToken(
        Guid eventId,
        ISessionTokenService tokenService,
        HttpContext httpContext)
    {
        var adminUserId = httpContext.User.GetUserId(); // From existing auth system
        var result = await tokenService.GenerateSessionTokenAsync(eventId, adminUserId);

        if (!result.IsSuccess)
        {
            return Results.BadRequest(ApiResponse.Failure(result.Error));
        }

        return Results.Ok(ApiResponse.Success(result.Value, "Session token generated successfully"));
    }

    private static async Task<IResult> RevokeSessionToken(
        string token,
        string reason,
        ISessionTokenService tokenService,
        HttpContext httpContext)
    {
        var adminUserId = httpContext.User.GetUserId();
        var result = await tokenService.RevokeTokenAsync(token, adminUserId, reason);

        if (!result.IsSuccess)
        {
            return Results.BadRequest(ApiResponse.Failure(result.Error));
        }

        return Results.Ok(ApiResponse.Success(true, "Session token revoked"));
    }
}
```

```csharp
/// <summary>
/// Endpoint filter that validates session tokens on every request
/// Replaces role-based authorization for check-in endpoints
/// </summary>
public class SessionTokenValidationFilter : IEndpointFilter
{
    private readonly ISessionTokenService _tokenService;
    private readonly ILogger<SessionTokenValidationFilter> _logger;

    public SessionTokenValidationFilter(
        ISessionTokenService tokenService,
        ILogger<SessionTokenValidationFilter> logger)
    {
        _tokenService = tokenService;
        _logger = logger;
    }

    public async ValueTask<object?> InvokeAsync(
        EndpointFilterInvocationContext context,
        EndpointFilterDelegate next)
    {
        // Extract token from custom header
        var httpContext = context.HttpContext;
        if (!httpContext.Request.Headers.TryGetValue("X-CheckIn-Token", out var tokenHeader))
        {
            _logger.LogWarning("Missing X-CheckIn-Token header");
            return Results.Unauthorized();
        }

        var token = tokenHeader.ToString();

        // Extract eventId from route
        var eventId = context.GetArgument<Guid>(0); // First argument is eventId

        // Validate token
        var validationResult = await _tokenService.ValidateTokenAsync(token, eventId);

        if (!validationResult.IsSuccess || !validationResult.Value.IsValid)
        {
            _logger.LogWarning("Invalid session token for event {EventId}", eventId);
            return Results.Unauthorized();
        }

        // Token valid - proceed to endpoint
        return await next(context);
    }
}
```

## Security Implementation

### Session Token Security Measures

1. **Cryptographic Strength**: 512-bit random tokens generated with `RandomNumberGenerator.Create()`
2. **Event Scoping**: Token only valid for specific event ID
3. **Time-Based Expiration**: Default 12-hour expiration, configurable
4. **Revocation Support**: Admin can revoke compromised tokens
5. **Audit Logging**: All token generation, usage, and revocation logged
6. **Cache-Based Validation**: Fast token validation using memory cache
7. **Rate Limiting**: Prevent token brute-force attempts
8. **IP Logging**: Track IP addresses using session tokens

## Performance Optimization Strategy

### Database Query Optimization
1. **Strategic Indexing**:
   - EventAttendees: (EventId, RegistrationStatus)
   - CheckIns: (EventId, CheckInTime DESC)
   - CheckInSessionTokens: (Token), (EventId, IsActive)
   - EventAttendees: (UserId) for user lookups
   - EventAttendees: Partial indexes on FirstTime and Waiver status

2. **Query Patterns**:
   - Use AsNoTracking() for read-only operations
   - Include navigation properties strategically
   - Paginate large attendee lists
   - Project to DTOs to avoid over-fetching

3. **Capacity Calculations**:
   ```sql
   -- Optimized capacity query with single DB hit
   SELECT
       e."Capacity" as "TotalCapacity",
       COUNT(ci."Id") as "CheckedInCount",
       (e."Capacity" - COUNT(ci."Id")) as "AvailableSpots",
       COUNT(ea."Id") FILTER (WHERE ea."RegistrationStatus" = 'waitlist') as "WaitlistCount"
   FROM "Events" e
   LEFT JOIN "EventAttendees" ea ON e."Id" = ea."EventId"
   LEFT JOIN "CheckIns" ci ON ea."Id" = ci."EventAttendeeId"
   WHERE e."Id" = @eventId
   GROUP BY e."Id", e."Capacity";
   ```

### Caching Strategy
```csharp
/// <summary>
/// Cache configuration for CheckIn system performance
/// </summary>
public class CacheService
{
    private readonly IMemoryCache _cache;
    private readonly ILogger<CacheService> _logger;

    // Cache keys
    private const string EVENT_CAPACITY_KEY = "event_capacity_{0}";
    private const string ATTENDEE_LIST_KEY = "attendee_list_{0}_{1}_{2}"; // eventId_page_search
    private const string EVENT_DASHBOARD_KEY = "event_dashboard_{0}";
    private const string SESSION_TOKEN_KEY = "checkin_token_{0}"; // CRITICAL for fast token validation

    // Cache durations
    private static readonly TimeSpan CAPACITY_CACHE_DURATION = TimeSpan.FromMinutes(2);
    private static readonly TimeSpan ATTENDEE_CACHE_DURATION = TimeSpan.FromMinutes(5);
    private static readonly TimeSpan DASHBOARD_CACHE_DURATION = TimeSpan.FromMinutes(1);
    private static readonly TimeSpan TOKEN_CACHE_DURATION = TimeSpan.FromHours(12);

    public async Task<T> GetOrSetAsync<T>(string key, Func<Task<T>> getItem, TimeSpan duration)
    {
        if (_cache.TryGetValue(key, out T cachedValue))
        {
            return cachedValue;
        }

        var item = await getItem();
        _cache.Set(key, item, duration);
        return item;
    }

    public void InvalidateEventCaches(Guid eventId)
    {
```

## Offline Capability Implementation

### Sync Queue Management
```csharp
/// <summary>
/// Offline sync service implementation
/// Handles network interruptions and data conflicts
/// NOTE: Session token included in sync requests for authorization
/// </summary>
public class SyncService : ISyncService
{
    public async Task<Result<SyncResponse>> ProcessOfflineSyncAsync(
        SyncRequest request,
        string sessionToken, // Required for kiosk mode authorization
        CancellationToken cancellationToken = default)
    {
        // Validate session token before processing sync
        var tokenValidation = await _tokenService.ValidateTokenAsync(
            sessionToken,
            Guid.Parse(request.EventId));

        if (!tokenValidation.IsSuccess || !tokenValidation.Value.IsValid)
        {
            return Result<SyncResponse>.Failure("Invalid or expired session token");
        }

        var conflicts = new List<SyncConflict>();
        var processedCount = 0;
        var updatedAttendees = new List<CheckInAttendee>();

        try
        {
            foreach (var pendingCheckIn in request.PendingCheckIns)
            {
                var result = await ProcessPendingCheckInAsync(pendingCheckIn, sessionToken, cancellationToken);

                if (result.IsSuccess)
                {
                    processedCount++;
                    // Add to updated attendees list
                }
                else
                {
                    // Check for conflicts
                    var conflict = await AnalyzeConflictAsync(pendingCheckIn, result.Error);
                    if (conflict != null)
                    {
                        conflicts.Add(conflict);
                    }
                }
            }

            return Result<SyncResponse>.Success(new SyncResponse
            {
                Success = true,
                ProcessedCount = processedCount,
                Conflicts = conflicts,
                UpdatedAttendees = updatedAttendees,
                NewSyncTimestamp = DateTime.UtcNow.ToString("O")
            });
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error during sync processing");
            return Result<SyncResponse>.Failure("Sync processing failed");
        }
    }

    private async Task<SyncConflict?> AnalyzeConflictAsync(PendingCheckIn pending, string error)
    {
        // Check for duplicate check-in
        var existingCheckIn = await _context.CheckIns
            .Include(ci => ci.EventAttendee)
            .FirstOrDefaultAsync(ci => ci.EventAttendeeId == Guid.Parse(pending.AttendeeId));

        if (existingCheckIn != null)
        {
            return new SyncConflict
            {
                LocalId = pending.LocalId,
                AttendeeId = pending.AttendeeId,
                ConflictType = "duplicate_checkin",
                ServerData = new {
                    checkInTime = existingCheckIn.CheckInTime,
                    sessionToken = existingCheckIn.SessionToken // Track which kiosk did original check-in
                },
                LocalData = new {
                    checkInTime = pending.CheckInTime,
                    sessionToken = pending.SessionToken
                },
                Resolution = DateTime.Parse(pending.CheckInTime) < existingCheckIn.CheckInTime ?
                    "auto_resolved" : "manual_required",
                Message = "Attendee was already checked in by another kiosk station"
            };
        }

        return null;
    }
}
```

### Conflict Resolution Strategy
1. **Duplicate Check-ins**: Preserve earliest timestamp, log both kiosk stations
2. **Capacity Conflicts**: Require manual admin approval via admin panel
3. **Attendee Not Found**: Flag for manual review and possible registration
4. **Token Expiration**: Clear error message instructing volunteer to request new link from admin

```csharp
/// <summary>
/// CheckIn system health monitoring
/// Includes session token system monitoring

public class CheckInHealthCheck : IHealthCheck
{
    public async Task<HealthCheckResult> CheckHealthAsync(HealthCheckContext context, CancellationToken cancellationToken = default)
    {
        try
        {
            // Test database connectivity
            var canConnect = await _context.Database.CanConnectAsync(cancellationToken);
            if (!canConnect)
            {
                return HealthCheckResult.Unhealthy("Database connection failed");
            }

            // Check for failed sync operations
            var failedSyncCount = await _context.OfflineSyncQueues
                .CountAsync(q => q.SyncStatus == "failed" && q.RetryCount >= 5, cancellationToken);

            // Check cache availability
            var cacheWorking = _cache.Get("health_check_key") != null;

            // Check active session tokens
            var activeTokenCount = await _context.CheckInSessionTokens
                .CountAsync(t => t.IsActive && t.ExpiresAt > DateTime.UtcNow, cancellationToken);

            var data = new Dictionary<string, object>
            {
                ["database_connected"] = canConnect,
                ["failed_sync_operations"] = failedSyncCount,
                ["cache_available"] = cacheWorking,
                ["active_session_tokens"] = activeTokenCount
            };

            if (failedSyncCount > 10)
            {
                return HealthCheckResult.Degraded("High number of failed sync operations", null, data);
            }

            return HealthCheckResult.Healthy("CheckIn system operational", data);
        }
        catch (Exception ex)
        {
            return HealthCheckResult.Unhealthy("CheckIn health check failed", ex);
        }
    }
}
```
